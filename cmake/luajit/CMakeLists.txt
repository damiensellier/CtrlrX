# ============================================================================
# CtrlrX: build the vendored LuaJIT from source.
#
# FORK-LOCAL FILE. Upstream LuaJIT ships only Makefile and src/msvcbuild.bat
# and has no CMake build; this file has no upstream counterpart. It lives here
# rather than inside Source/Misc/luajit/ so that the vendored tree stays
# byte-identical to upstream and can be replaced wholesale on an upgrade
# without deleting the build -- see Doc/LUAJIT.md.
#
# The source lists below are transcribed from the vendored src/Makefile and can
# drift from it silently. check_makefile_drift.py (next to this file, run by CI)
# compares the two and fails on any difference.
#
# Configurable standalone as well as via the parent project:
#   cmake -S cmake/luajit -B <dir> && cmake --build <dir>
# Point CTRLRX_LUAJIT_SOURCE_DIR at a different LuaJIT checkout to build that
# one instead of the vendored tree.
#
# Reproduces LuaJIT's two-stage bootstrap:
#
#   minilua                                       host tool, arch-agnostic
#   minilua genversion.lua                      -> luajit.h, luajit_relver.txt
#   minilua dynasm.lua <flags> vm_<arch>.dasc   -> host/buildvm_arch.h
#   buildvm = host/buildvm*.c + buildvm_arch.h    host exe, target-aware
#   buildvm -m elfasm|machasm|peobj             -> lj_vm.S / .s / .obj
#   buildvm -m bcdef|ffdef|libdef|recdef        -> lj_*def.h
#   buildvm -m folddef                          -> lj_folddef.h
#   compile LJCORE_C + lj_vm                    -> libluajit_<slice>
#   lipo (Apple, >1 arch)                       -> libluajit
#
# Only minilua is shared between architectures. Everything from
# buildvm_arch.h down is per-target-arch, which is why a macOS universal
# build runs the chain once per slice and lipo's the results.
#
# Generated files go to ${CMAKE_CURRENT_BINARY_DIR}/gen/<slice>/, never into
# src/ -- two slices would otherwise overwrite each other's lj_bcdef.h.
# ============================================================================

cmake_minimum_required(VERSION 3.25)

# Standalone configure only. As a subdirectory of CtrlrX the parent project has
# already run project(), which enables C -- calling it again would reset the
# toolchain settings the parent established.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LuaJIT C)
endif()

option(CTRLRX_LUAJIT_DEBUG "Build the vendored LuaJIT unoptimised (for debugging LuaJIT itself)" OFF)

set(CTRLRX_LUAJIT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../Source/Misc/luajit"
    CACHE PATH "Location of the vendored LuaJIT source tree")

# The interpreter core is assembly generated by buildvm. MSVC has no assembler
# for it, so there we consume a ready-made PE object instead. enable_language()
# only works at directory scope, so it cannot live in _luajit_add_slice().
if(NOT WIN32)
    enable_language(ASM)
endif()

# ----------------------------------------------------------------------------
# Escape hatch: system LuaJIT (distro packagers, or A/B-ing against the vendored
# build). Not the default -- it means this platform runs a different LuaJIT from
# the others.
# ----------------------------------------------------------------------------
if(CTRLRX_LUAJIT_USE_SYSTEM)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SYSLUAJIT REQUIRED luajit)
    add_library(libluajit INTERFACE)
    target_include_directories(libluajit INTERFACE ${SYSLUAJIT_INCLUDE_DIRS})
    target_link_libraries(libluajit INTERFACE ${SYSLUAJIT_LIBRARIES})
    target_link_directories(libluajit INTERFACE ${SYSLUAJIT_LIBRARY_DIRS})

    # Keep the `luajit_all` target name valid on both paths, so that building it
    # does not depend on how LuaJIT was sourced. There is nothing to build here.
    # (no semicolons in the message -- CMake would split it into two commands)
    add_custom_target(luajit_all
        COMMAND ${CMAKE_COMMAND} -E echo
            "LuaJIT: using the system library, nothing to build.")

    message(STATUS "LuaJIT: using system ${SYSLUAJIT_VERSION} (CTRLRX_LUAJIT_USE_SYSTEM=ON)")
    return()
endif()

# ----------------------------------------------------------------------------
# Stale-file guard.
#
# `#include "lj_bcdef.h"` from a file in src/ searches src/ first, ahead of any
# -I. A leftover generated file there silently shadows the one we generate,
# producing a baffling miscompile rather than an error. Fail loudly instead.
#
# The guard also rejects compiled artefacts (libluajit.a, lua51.lib, *.o) 
# so that the tree stays byte-identical to upstream
# ----------------------------------------------------------------------------
set(_lj_generated_in_tree
    "${LJ_SRC}/luajit.h"
    "${LJ_SRC}/luajit_relver.txt"
    "${LJ_SRC}/lj_bcdef.h"
    "${LJ_SRC}/lj_ffdef.h"
    "${LJ_SRC}/lj_libdef.h"
    "${LJ_SRC}/lj_recdef.h"
    "${LJ_SRC}/lj_folddef.h"
    "${LJ_SRC}/lj_vm.S"
    "${LJ_SRC}/lj_vm.s"
    "${LJ_SRC}/host/buildvm_arch.h"
    "${LJ_SRC}/jit/vmdef.lua")

# Compiled output.
file(GLOB _lj_binaries_in_tree
    "${LJ_SRC}/*.o" "${LJ_SRC}/*.obj" "${LJ_SRC}/*.a" "${LJ_SRC}/*.lib"
    "${LJ_SRC}/*.dll" "${LJ_SRC}/*.so" "${LJ_SRC}/*.dylib"
    "${LJ_SRC}/host/*.o" "${LJ_SRC}/host/*.obj")

set(_lj_stale "")
foreach(_f IN LISTS _lj_generated_in_tree _lj_binaries_in_tree)
    if(EXISTS "${_f}")
        # Relative to LJ_DIR, not CMAKE_SOURCE_DIR: the top level differs
        # between a standalone configure and one under CtrlrX.
        file(RELATIVE_PATH _rel "${LJ_DIR}" "${_f}")
        string(APPEND _lj_stale "\n    ${_rel}")
    endif()
endforeach()

if(_lj_stale)
    message(FATAL_ERROR
        "LuaJIT: build output found in the vendored source tree ${LJ_DIR}:"
        "${_lj_stale}\n\n"
        "Generated headers shadow the ones CMake generates, because a quoted "
        "#include searches the including file's own directory first."
        "Compiled libraries and objects do not shadow anything, but they do"
        "not belong in a tree that must stay byte-identical to upstream.\n\n"
        "See Doc/LUAJIT.md."
        "\n\nRemove them with:\n\n"
        "    git clean -xdf Source/Misc/luajit\n")
endif()

# ----------------------------------------------------------------------------
# luajit.h -- architecture-independent, generated once.
#
# genversion.lua substitutes ROLLING in luajit_rolling.h with the digits it
# finds in luajit_relver.txt. Upstream fills that from `git show -s --format=%ct`
# in the LuaJIT checkout, but Source/Misc/luajit is a plain directory here, not
# a submodule, so that would report CtrlrX's commit time and produce a
# plausible-looking but wrong LuaJIT version. We feed it the shipped .relver
# ($Format:%ct$, an unexpanded git-archive placeholder); genversion finds no
# digits and keeps "LuaJIT 2.1.ROLLING", which is what macOS and Windows have
# been shipping all along. CtrlrAbout.cpp displays this string.
# ----------------------------------------------------------------------------
set(LJ_GEN_COMMON "${LJ_GEN}/common")
file(MAKE_DIRECTORY "${LJ_GEN_COMMON}")

add_custom_command(
    OUTPUT "${LJ_GEN_COMMON}/luajit.h"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${LJ_DIR}/.relver" "${LJ_GEN_COMMON}/luajit_relver.txt"
    COMMAND luajit_minilua "${LJ_SRC}/host/genversion.lua"
        "${LJ_SRC}/luajit_rolling.h"
        "${LJ_GEN_COMMON}/luajit_relver.txt"
        "${LJ_GEN_COMMON}/luajit.h"
    DEPENDS luajit_minilua "${LJ_SRC}/luajit_rolling.h" "${LJ_DIR}/.relver"
    VERBATIM
    COMMENT "LuaJIT: generating luajit.h")

add_custom_target(luajit_version DEPENDS "${LJ_GEN_COMMON}/luajit.h")
set_target_properties(luajit_version PROPERTIES FOLDER "LuaJIT")

# ----------------------------------------------------------------------------
# Architecture and DynASM flag detection.
#
# On Clang/GCC this mirrors src/Makefile:236-267 and 405-485 exactly: dump the
# preprocessor's view of lj_arch.h and derive everything from it. That is what
# keeps us honest across LuaJIT upgrades -- if upstream flips a default (GC64,
# PAUTH, BRANCH_TRACK) the probe picks it up with no edit here.
#
# MSVC cannot dump macros, so it gets a table transcribed from
# msvcbuild.bat:46-64 -- which is what upstream itself does on Windows.
# ----------------------------------------------------------------------------
function(_luajit_detect slice out_ljarch out_dasmarch out_dasmflags out_defs)
    set(_archflag "")
    if(APPLE AND NOT slice STREQUAL "native")
        set(_archflag "-arch" "${slice}")
    endif()

    if(MSVC)
        # No -dM equivalent. Table from msvcbuild.bat.
        set(_p "${CMAKE_VS_PLATFORM_NAME}")
        if(_p STREQUAL "")
            set(_p "${CMAKE_SYSTEM_PROCESSOR}")
        endif()
        string(TOLOWER "${_p}" _p)

        if(_p MATCHES "arm64|aarch64")
            set(_ljarch arm64)
            set(_dasmarch arm64)
            set(_flags -D WIN -D JIT -D FFI -D ENDIAN_LE -D FPU -D P64)
            set(_defs LUAJIT_TARGET=LUAJIT_ARCH_ARM64)
        elseif(_p MATCHES "x64|amd64|x86_64" OR CMAKE_SIZEOF_VOID_P EQUAL 8)
            set(_ljarch x64)
            set(_dasmarch x64)
            set(_flags -D WIN -D JIT -D FFI -D ENDIAN_LE -D FPU -D P64)
            set(_defs "")
        else()
            set(_ljarch x86)
            set(_dasmarch x86)
            set(_flags -D WIN -D JIT -D FFI -D ENDIAN_LE -D FPU)
            set(_defs "")
        endif()

        set(${out_ljarch} "${_ljarch}" PARENT_SCOPE)
        set(${out_dasmarch} "${_dasmarch}" PARENT_SCOPE)
        set(${out_dasmflags} "${_flags}" PARENT_SCOPE)
        set(${out_defs} "${_defs}" PARENT_SCOPE)
        return()
    endif()

    execute_process(
        COMMAND ${CMAKE_C_COMPILER} ${_archflag} -I "${LJ_SRC}" -E -dM "${LJ_SRC}/lj_arch.h"
        WORKING_DIRECTORY "${LJ_SRC}"
        OUTPUT_VARIABLE _dump
        ERROR_VARIABLE _err
        RESULT_VARIABLE _rc)

    if(NOT _rc EQUAL 0)
        message(FATAL_ERROR
            "LuaJIT: failed to probe lj_arch.h for slice '${slice}'.\n"
            "Command: ${CMAKE_C_COMPILER} ${_archflag} -E -dM lj_arch.h\n"
            "Exit code: ${_rc}\n${_err}")
    endif()

    # Target architecture (Makefile:237-267).
    if(_dump MATCHES "LJ_TARGET_X64 ")
        set(_ljarch x64)
    elseif(_dump MATCHES "LJ_TARGET_X86 ")
        set(_ljarch x86)
    elseif(_dump MATCHES "LJ_TARGET_ARM64 ")
        set(_ljarch arm64)
    elseif(_dump MATCHES "LJ_TARGET_ARM ")
        set(_ljarch arm)
    else()
        message(FATAL_ERROR
            "LuaJIT: unsupported target architecture for slice '${slice}'. "
            "CtrlrX builds the vendored LuaJIT for x86/x64/arm/arm64 only. "
            "To use a system LuaJIT instead, configure with "
            "-DCTRLRX_LUAJIT_USE_SYSTEM=ON.")
    endif()

    set(_dasmarch "${_ljarch}")
    set(_flags "")
    set(_defs "")

    # DASM_AFLAGS (Makefile:409-453).
    if(_dump MATCHES "LJ_LE 1")
        list(APPEND _flags -D ENDIAN_LE)
    else()
        list(APPEND _flags -D ENDIAN_BE)
    endif()
    if(_dump MATCHES "LJ_ARCH_BITS 64")
        list(APPEND _flags -D P64)
    endif()
    if(_dump MATCHES "LJ_HASJIT 1")
        list(APPEND _flags -D JIT)
    endif()
    if(_dump MATCHES "LJ_HASFFI 1")
        list(APPEND _flags -D FFI)
    endif()
    if(_dump MATCHES "LJ_DUALNUM 1")
        list(APPEND _flags -D DUALNUM)
    endif()
    if(_dump MATCHES "LJ_ARCH_HASFPU 1")
        list(APPEND _flags -D FPU)
        list(APPEND _defs LJ_ARCH_HASFPU=1)
    else()
        list(APPEND _defs LJ_ARCH_HASFPU=0)
    endif()
    if(_dump MATCHES "LJ_ABI_SOFTFP 1")
        list(APPEND _defs LJ_ABI_SOFTFP=1)
    else()
        list(APPEND _flags -D HFABI)
        list(APPEND _defs LJ_ABI_SOFTFP=0)
    endif()
    if(_dump MATCHES "LJ_NO_UNWIND 1")
        list(APPEND _flags -D NO_UNWIND)
        list(APPEND _defs LUAJIT_NO_UNWIND)
    endif()
    if(_dump MATCHES "LJ_ABI_PAUTH 1")
        list(APPEND _flags -D PAUTH)
        list(APPEND _defs LJ_ABI_PAUTH=1)
    endif()
    if(_dump MATCHES "LJ_ABI_BRANCH_TRACK 1")
        list(APPEND _flags -D BRANCH_TRACK)
        list(APPEND _defs LJ_ABI_BRANCH_TRACK=1)
    endif()
    if(_dump MATCHES "LJ_ABI_SHADOW_STACK 1")
        list(APPEND _flags -D SHADOW_STACK)
        list(APPEND _defs LJ_ABI_SHADOW_STACK=1)
    endif()

    # -D VER=<n> from LJ_ARCH_VERSION (Makefile:454).
    if(_dump MATCHES "LJ_ARCH_VERSION ([0-9]+)")
        list(APPEND _flags -D "VER=${CMAKE_MATCH_1}")
    else()
        list(APPEND _flags -D VER=)
    endif()

    if(WIN32)
        list(APPEND _flags -D WIN)
    endif()

    # x64 without FR2 (i.e. GC64 disabled) uses the x86 VM description
    # (Makefile:458-462).
    if(_ljarch STREQUAL "x64" AND NOT _dump MATCHES "LJ_FR2 1")
        set(_dasmarch x86)
    endif()

    list(APPEND _defs "LUAJIT_TARGET=LUAJIT_ARCH_${_ljarch}")

    set(${out_ljarch} "${_ljarch}" PARENT_SCOPE)
    set(${out_dasmarch} "${_dasmarch}" PARENT_SCOPE)
    set(${out_dasmflags} "${_flags}" PARENT_SCOPE)
    set(${out_defs} "${_defs}" PARENT_SCOPE)
endfunction()

# Target-side compile flags that are not derived from the arch probe.

include(CheckCCompilerFlag)

set(LJ_TARGET_XCFLAGS "")
set(LJ_TARGET_XDEFS "")

if(NOT WIN32)
    check_c_compiler_flag(-fno-stack-protector LJ_HAS_NO_STACK_PROTECTOR)
    if(LJ_HAS_NO_STACK_PROTECTOR)
        list(APPEND LJ_TARGET_XCFLAGS -fno-stack-protector)
    endif()
endif()

if(APPLE)
    # Makefile:322 sets this unconditionally for Darwin.
    list(APPEND LJ_TARGET_XDEFS LUAJIT_UNWIND_EXTERNAL)
elseif(UNIX)
    # Makefile:339-344 probes whether the toolchain always emits unwind tables,
    # by compiling a snippet and looking for eh_frame/__unwind_info in the
    # object. Matters for exception interop -- luabind throws through Lua.
    set(_probe_c "${CMAKE_CURRENT_BINARY_DIR}/lj_unwind_probe.c")
    set(_probe_o "${CMAKE_CURRENT_BINARY_DIR}/lj_unwind_probe.o")
    file(WRITE "${_probe_c}" "extern void b(void);int a(void){b();return 0;}\n")
    execute_process(
        COMMAND ${CMAKE_C_COMPILER} -c "${_probe_c}" -o "${_probe_o}"
        RESULT_VARIABLE _probe_rc
        OUTPUT_QUIET ERROR_QUIET)
    if(_probe_rc EQUAL 0 AND EXISTS "${_probe_o}")
        file(READ "${_probe_o}" _probe_hex HEX)
        # "eh_frame" and "__unwind_info" as hex byte strings.
        string(FIND "${_probe_hex}" "65685f6672616d65" _has_ehframe)
        string(FIND "${_probe_hex}" "5f5f756e77696e645f696e666f" _has_unwindinfo)
        if(NOT _has_ehframe EQUAL -1 OR NOT _has_unwindinfo EQUAL -1)
            list(APPEND LJ_TARGET_XDEFS LUAJIT_UNWIND_EXTERNAL)
        endif()
    endif()
    file(REMOVE "${_probe_c}" "${_probe_o}")
endif()

# ----------------------------------------------------------------------------
# One architecture slice: buildvm_arch.h -> buildvm -> lj_vm + lj_*def.h ->
# libluajit_<slice>.
# ----------------------------------------------------------------------------
function(_luajit_add_slice slice out_target)
    _luajit_detect("${slice}" _ljarch _dasmarch _dasmflags _archdefs)

    set(_gen "${LJ_GEN}/${slice}")
    file(MAKE_DIRECTORY "${_gen}/host")
    file(MAKE_DIRECTORY "${_gen}/jit")

    message(STATUS "LuaJIT: slice '${slice}' -> ${_ljarch} (VM ${_dasmarch})")

    # -- buildvm_arch.h ------------------------------------------------------
    add_custom_command(
        OUTPUT "${_gen}/host/buildvm_arch.h"
        COMMAND luajit_minilua "${LJ_DASM_DIR}/dynasm.lua" ${_dasmflags}
            -o "${_gen}/host/buildvm_arch.h" "${LJ_SRC}/vm_${_dasmarch}.dasc"
        DEPENDS luajit_minilua
            "${LJ_SRC}/vm_${_dasmarch}.dasc"
            "${LJ_DASM_DIR}/dynasm.lua"
            "${LJ_SRC}/lj_arch.h"
        WORKING_DIRECTORY "${LJ_SRC}"
        VERBATIM
        COMMENT "LuaJIT[${slice}]: dynasm -> buildvm_arch.h")

    # -- buildvm (host executable, target-aware) -----------------------------
    set(_buildvm "luajit_buildvm_${slice}")
    set(_buildvm_src "")
    foreach(_f IN LISTS LJ_BUILDVM_C)
        list(APPEND _buildvm_src "${LJ_SRC}/${_f}")
    endforeach()

    add_executable(${_buildvm} ${_buildvm_src} "${_gen}/host/buildvm_arch.h")
    add_dependencies(${_buildvm} luajit_version)
    target_include_directories(${_buildvm} PRIVATE
        "${_gen}/host"
        "${LJ_GEN_COMMON}"
        "${LJ_SRC}"
        "${LJ_DASM_DIR}")
    target_compile_definitions(${_buildvm} PRIVATE ${_archdefs})

    if(APPLE)
        set_target_properties(${_buildvm} PROPERTIES OSX_ARCHITECTURES "${LJ_HOST_ARCH}")
    endif()
    if(MSVC)
        target_compile_definitions(${_buildvm} PRIVATE _CRT_SECURE_NO_DEPRECATE)
    endif()
    set_target_properties(${_buildvm} PROPERTIES FOLDER "LuaJIT")

    # -- lj_vm ---------------------------------------------------------------
    # Mode per target system (Makefile:509-580): ELF assembly on Linux, Mach-O
    # assembly on Apple, and a ready-made PE object on Windows because MSVC has
    # no assembler for this.
    if(WIN32)
        set(_vm_out "${_gen}/lj_vm.obj")
        set(_vm_mode peobj)
    elseif(APPLE)
        set(_vm_out "${_gen}/lj_vm.s")
        set(_vm_mode machasm)
    else()
        set(_vm_out "${_gen}/lj_vm.S")
        set(_vm_mode elfasm)
    endif()

    add_custom_command(
        OUTPUT "${_vm_out}"
        COMMAND ${_buildvm} -m ${_vm_mode} -o "${_vm_out}"
        DEPENDS ${_buildvm}
        VERBATIM
        COMMENT "LuaJIT[${slice}]: buildvm -m ${_vm_mode}")

    # -- generated headers ---------------------------------------------------
    set(_gen_headers "")
    foreach(_mode bcdef ffdef libdef recdef)
        add_custom_command(
            OUTPUT "${_gen}/lj_${_mode}.h"
            COMMAND ${_buildvm} -m ${_mode} -o "${_gen}/lj_${_mode}.h" ${LJ_LIB_C_ABS}
            DEPENDS ${_buildvm} ${LJ_LIB_C_ABS}
            VERBATIM
            COMMENT "LuaJIT[${slice}]: buildvm -m ${_mode}")
        list(APPEND _gen_headers "${_gen}/lj_${_mode}.h")
    endforeach()

    add_custom_command(
        OUTPUT "${_gen}/lj_folddef.h"
        COMMAND ${_buildvm} -m folddef -o "${_gen}/lj_folddef.h" "${LJ_SRC}/lj_opt_fold.c"
        DEPENDS ${_buildvm} "${LJ_SRC}/lj_opt_fold.c"
        VERBATIM
        COMMENT "LuaJIT[${slice}]: buildvm -m folddef")
    list(APPEND _gen_headers "${_gen}/lj_folddef.h")

    # -- the library ---------------------------------------------------------
    set(_lib "libluajit_${slice}")
    set(_core_src "")
    foreach(_f IN LISTS LJ_CORE_C)
        list(APPEND _core_src "${LJ_SRC}/${_f}")
    endforeach()

    add_library(${_lib} STATIC ${_core_src} ${_gen_headers} "${_vm_out}")

    if(WIN32)
        # A prebuilt PE object, not something to assemble.
        set_source_files_properties("${_vm_out}" PROPERTIES
            EXTERNAL_OBJECT TRUE GENERATED TRUE)
    else()
        set_source_files_properties("${_vm_out}" PROPERTIES
            LANGUAGE ASM GENERATED TRUE)
    endif()

    target_include_directories(${_lib} PUBLIC
        "${_gen}"
        "${LJ_GEN_COMMON}"
        "${LJ_SRC}")
    target_compile_definitions(${_lib} PRIVATE ${_archdefs} ${LJ_TARGET_XDEFS})
    target_compile_options(${_lib} PRIVATE ${LJ_TARGET_XCFLAGS})

    if(MSVC)
        # msvcbuild.bat sets this for every translation unit; LuaJIT's core uses
        # fopen/getenv/strcpy throughout and would otherwise emit C4996 for each.
        target_compile_definitions(${_lib} PRIVATE _CRT_SECURE_NO_DEPRECATE)
    endif()
    add_dependencies(${_lib} luajit_version)

    # Linked into shared plugin binaries (VST3/AU).
    set_target_properties(${_lib} PROPERTIES
        POSITION_INDEPENDENT_CODE TRUE
        OUTPUT_NAME "luajit_${slice}")

    if(APPLE AND NOT slice STREQUAL "native")
        set_target_properties(${_lib} PROPERTIES OSX_ARCHITECTURES "${slice}")
    endif()

    # LuaJIT is always built optimised, as upstream does (CCOPT= -O2
    # -fomit-frame-pointer), regardless of CMAKE_BUILD_TYPE. A Debug CtrlrX
    # still runs panel Lua at normal speed, and we stay on the code paths
    # upstream actually tests. Debug info still follows the build type.
    if(NOT CTRLRX_LUAJIT_DEBUG)
        if(MSVC)
            target_compile_options(${_lib} PRIVATE /O2)
        else()
            target_compile_options(${_lib} PRIVATE -O2 -fomit-frame-pointer)
        endif()
    endif()

    set_target_properties(${_lib} PROPERTIES FOLDER "LuaJIT")
    set(${out_target} "${_lib}" PARENT_SCOPE)
endfunction()

# ----------------------------------------------------------------------------
# Build one slice per target architecture.
# ----------------------------------------------------------------------------
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
    set(LJ_SLICES ${CMAKE_OSX_ARCHITECTURES})
else()
    set(LJ_SLICES "native")
endif()

set(LJ_SLICE_TARGETS "")
foreach(_slice IN LISTS LJ_SLICES)
    _luajit_add_slice("${_slice}" _slice_target)
    list(APPEND LJ_SLICE_TARGETS "${_slice_target}")
endforeach()

list(LENGTH LJ_SLICE_TARGETS LJ_SLICE_COUNT)

# ----------------------------------------------------------------------------
# Publish `libluajit`.
#
# Single architecture: alias the slice. Multiple (macOS universal): the
# consumer links a universal binary, so each arch link needs its arch present
# in the archive -- two thin archives will not do, they have to be merged with
# lipo. We build a real STATIC target and overwrite its archive with the fat
# one POST_BUILD, which keeps libluajit a first-class target with correct
# build ordering.
# ----------------------------------------------------------------------------
if(LJ_SLICE_COUNT EQUAL 1)
    add_library(libluajit INTERFACE)
    target_link_libraries(libluajit INTERFACE ${LJ_SLICE_TARGETS})
else()
    if(NOT APPLE)
        message(FATAL_ERROR
            "LuaJIT: multiple architecture slices are only supported on Apple "
            "platforms (via lipo). Got: ${LJ_SLICES}")
    endif()

    set(_stub "${CMAKE_CURRENT_BINARY_DIR}/luajit_fat_stub.c")
    file(WRITE "${_stub}" "/* placeholder; replaced by the lipo'd archive */\n"
        "int luajit_fat_stub(void);\nint luajit_fat_stub(void){return 0;}\n")

    add_library(libluajit STATIC "${_stub}")
    set_target_properties(libluajit PROPERTIES
        OUTPUT_NAME "luajit"
        OSX_ARCHITECTURES "${LJ_HOST_ARCH}")

    set(_slice_files "")
    foreach(_t IN LISTS LJ_SLICE_TARGETS)
        list(APPEND _slice_files "$<TARGET_FILE:${_t}>")
        add_dependencies(libluajit ${_t})
    endforeach()

    add_custom_command(TARGET libluajit POST_BUILD
        COMMAND lipo -create ${_slice_files} -output "$<TARGET_FILE:libluajit>"
        COMMAND lipo -info "$<TARGET_FILE:libluajit>"
        VERBATIM
        COMMENT "LuaJIT: lipo ${LJ_SLICES} -> universal libluajit")

    target_include_directories(libluajit PUBLIC
        "${LJ_GEN}/${LJ_HOST_ARCH}"
        "${LJ_GEN_COMMON}"
        "${LJ_SRC}")
endif()

set_target_properties(libluajit PROPERTIES FOLDER "LuaJIT")

# `libluajit` is an INTERFACE library on single-architecture platforms, and an
# INTERFACE library is not a buildable target -- `cmake --build --target
# libluajit` fails with "unknown target". Provide a stable aggregate so LuaJIT
# can be built on its own, which is what CI does to isolate bootstrap failures
# from the rest of the CtrlrX build.
add_custom_target(luajit_all)
add_dependencies(luajit_all ${LJ_SLICE_TARGETS})
if(NOT LJ_SLICE_COUNT EQUAL 1)
    add_dependencies(luajit_all libluajit)
endif()
set_target_properties(luajit_all PROPERTIES FOLDER "LuaJIT")

message(STATUS "LuaJIT: building vendored source, slices: ${LJ_SLICES}")
